cmake_minimum_required(VERSION 4.2.0)
project(Lenerith)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_STANDARD 23)
set(PROJECT_INCLUDE "${CMAKE_CURRENT_LIST_DIR}/include")
set(PROJECT_LIB "${CMAKE_CURRENT_LIST_DIR}/lib")

file(GLOB SRC_FILES src/*.cpp)
add_executable(${PROJECT_NAME} ${SRC_FILES})

target_compile_definitions(${PROJECT_NAME} PRIVATE
    $<$<CONFIG:Debug>:DEBUG>
    $<$<CONFIG:Release>:NDEBUG>
)

target_include_directories(${PROJECT_NAME} PRIVATE ${PROJECT_INCLUDE})
target_link_directories(${PROJECT_NAME} PRIVATE ${PROJECT_LIB})

if(WIN32)
    target_link_directories(${PROJECT_NAME} PRIVATE ${PROJECT_LIB}/windows)
    target_link_libraries(${PROJECT_NAME} PRIVATE SDL3 SDL3_image winmm gdi32 opengl32)
    target_link_options(${PROJECT_NAME} PRIVATE
        $<$<CONFIG:Debug>:-fuse-ld=lld>
        $<$<CONFIG:Release>:-O3 -fuse-ld=lld -Wl,/SUBSYSTEM:WINDOWS -Wl,/ENTRY:mainCRTStartup>
    )
elseif(UNIX)
    target_link_directories(${PROJECT_NAME} PRIVATE ${PROJECT_LIB}/linux)
    target_link_libraries(${PROJECT_NAME} PRIVATE SDL3 SDL3_image)
    target_link_options(${PROJECT_NAME} PRIVATE
        $<$<CONFIG:Debug>:-Wall -Wextra>
        $<$<CONFIG:Release>:-O3 -Wall -Wextra>
    )
endif()
