include(FetchContent)

cmake_minimum_required(VERSION 4.2.0)
project(Lenerith)

set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED YES)
set(SDLTTF_VENDORED ON)

FetchContent_Declare(
  SDL3
  GIT_REPOSITORY https://github.com/libsdl-org/SDL.git
  GIT_TAG        main
)

FetchContent_Declare(
  SDL3_mixer
  GIT_REPOSITORY https://github.com/libsdl-org/SDL_mixer.git
  GIT_TAG        main
)

FetchContent_Declare(
  SDL3_image
  GIT_REPOSITORY https://github.com/libsdl-org/SDL_image.git
  GIT_TAG        main
)

FetchContent_Declare(
  SDL3_ttf
  GIT_REPOSITORY https://github.com/libsdl-org/SDL_ttf.git
  GIT_TAG        main
  GIT_SUBMODULES_RECURSE TRUE
)

FetchContent_MakeAvailable(SDL3 SDL3_mixer SDL3_image SDL3_ttf)

add_library(${PROJECT_NAME}
# Core
src/Core/Engine.cpp
src/Core/Renderer.cpp
src/Core/Window.cpp
src/Core/Config.cpp
src/Core/Audio.cpp
src/Core/Time.cpp
src/Core/Game.cpp
# Math
src/Math/Vector2.cpp
# Input
src/Input/Input.cpp
# Entity
src/Entity/Entity.cpp
# src/Entity/Player.cpp
src/Entity/Object.cpp
# Scene
src/Scene/Scene.cpp
src/Scene/SceneManager.cpp
# UI
# src/UI/UI.cpp
src/UI/Button.cpp
# Rest
src/Animation.cpp
src/Timer.cpp
src/Camera.cpp
src/TriggerBox.cpp
)

target_compile_definitions(${PROJECT_NAME} PRIVATE
    $<$<CONFIG:Debug>:DEBUG>
    $<$<CONFIG:Release>:NDEBUG>
)

set_target_properties(${PROJECT_NAME} PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY_DEBUG   ${CMAKE_BINARY_DIR}/Debug
    RUNTIME_OUTPUT_DIRECTORY_RELEASE ${CMAKE_BINARY_DIR}/Release
)

target_include_directories(${PROJECT_NAME} PRIVATE include)

target_link_libraries(${PROJECT_NAME} PRIVATE
    SDL3::SDL3
    SDL3_image::SDL3_image
    SDL3_mixer::SDL3_mixer
    SDL3_ttf::SDL3_ttf
)

if(WIN32)
  target_link_options(${PROJECT_NAME} PRIVATE
    $<$<CONFIG:Debug>:-fuse-ld=lld -Wall -Wextra>
    $<$<CONFIG:Release>:-O3 -fuse-ld=lld -Wall -Wextra>
  )
elseif(UNIX)
  target_link_options(${PROJECT_NAME} PRIVATE
    $<$<CONFIG:Debug>:-Wall -Wextra>
    $<$<CONFIG:Release>:-O3 -Wall -Wextra>
  )
endif()
